BRM_eth_init = {
	if = {
		limit = { NOT = { has_country_flag = BRM_eth_initialized } }
		set_country_flag = BRM_eth_initialized
		# Karen, Karenni, Mon and Arakanese insurgencies were already under way in 1949
		set_variable = { BRM_eth_karen = 2 } set_variable = { BRM_eth_karen_str = 8 }
		set_variable = { BRM_eth_karenni = 2 } set_variable = { BRM_eth_karenni_str = 4 }
		set_variable = { BRM_eth_mon = 2 } set_variable = { BRM_eth_mon_str = 4 }
		set_variable = { BRM_eth_arakan = 2 } set_variable = { BRM_eth_arakan_str = 3 }
		set_variable = { BRM_eth_shan = 0 } set_variable = { BRM_eth_shan_str = 0 }
		set_variable = { BRM_eth_kachin = 0 } set_variable = { BRM_eth_kachin_str = 0 }
		set_variable = { BRM_eth_chin = 0 } set_variable = { BRM_eth_chin_str = 0 }
		set_variable = { BRM_eth_wa = 0 } set_variable = { BRM_eth_wa_str = 0 }
		# later start dates (1980 bookmark) and older saves
		if = { limit = { date > 1959.5.21 } set_variable = { BRM_eth_shan = 2 } set_variable = { BRM_eth_shan_str = 6 } }
		if = { limit = { date > 1961.2.5 } set_variable = { BRM_eth_kachin = 2 } set_variable = { BRM_eth_kachin_str = 7 } set_variable = { BRM_eth_arakan_str = 2 } }
		if = { limit = { date > 1975.1.1 } set_variable = { BRM_eth_karen_str = 6 } set_variable = { BRM_eth_karenni_str = 3 } set_variable = { BRM_eth_mon_str = 3 } }
		if = { limit = { date > 1988.3.1 } set_variable = { BRM_eth_chin = 2 } set_variable = { BRM_eth_chin_str = 2 } }
		if = { limit = { date > 1989.4.17 } set_country_flag = BRM_eth_wa_active set_variable = { BRM_eth_wa = 1 } set_variable = { BRM_eth_wa_str = 8 } }
		add_dynamic_modifier = { modifier = BRM_ethnic_conflicts }
		BRM_eth_recalc = yes
	}
}
BRM_eth_recalc = {
	set_variable = { BRM_eth_wars = 0 }
	set_variable = { BRM_eth_ceasefires = 0 }
	set_variable = { BRM_eth_autonomies = 0 }
	if = { limit = { check_variable = { BRM_eth_karen = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_karen = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_karen_autonomy check_variable = { BRM_eth_karen = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	if = { limit = { check_variable = { BRM_eth_karenni = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_karenni = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_karenni_autonomy check_variable = { BRM_eth_karenni = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	if = { limit = { check_variable = { BRM_eth_mon = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_mon = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_mon_autonomy check_variable = { BRM_eth_mon = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	if = { limit = { check_variable = { BRM_eth_arakan = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_arakan = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_arakan_autonomy check_variable = { BRM_eth_arakan = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	if = { limit = { check_variable = { BRM_eth_shan = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_shan = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_shan_autonomy check_variable = { BRM_eth_shan = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	if = { limit = { check_variable = { BRM_eth_kachin = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_kachin = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_kachin_autonomy check_variable = { BRM_eth_kachin = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	if = { limit = { check_variable = { BRM_eth_chin = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_chin = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_chin_autonomy check_variable = { BRM_eth_chin = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	if = { limit = { check_variable = { BRM_eth_wa = 2 } } add_to_variable = { BRM_eth_wars = 1 } }
	if = { limit = { check_variable = { BRM_eth_wa = 1 } } add_to_variable = { BRM_eth_ceasefires = 1 } }
	if = { limit = { has_country_flag = BRM_eth_wa_autonomy check_variable = { BRM_eth_wa = 0 } } add_to_variable = { BRM_eth_autonomies = 1 } }
	# each armed conflict: -1% stability, -0.015 political power a day, -2% construction speed; each autonomy deal: +0.5% stability
	set_variable = { BRM_eth_stab = BRM_eth_wars }
	multiply_variable = { BRM_eth_stab = -0.01 }
	set_temp_variable = { brm_eth_t = BRM_eth_autonomies }
	multiply_temp_variable = { brm_eth_t = 0.005 }
	add_to_variable = { BRM_eth_stab = brm_eth_t }
	set_variable = { BRM_eth_pp = BRM_eth_wars }
	multiply_variable = { BRM_eth_pp = -0.015 }
	set_variable = { BRM_eth_build = BRM_eth_wars }
	multiply_variable = { BRM_eth_build = -0.02 }
	# money: ceasefire business (jade, timber, border trade) +0.01B each, counter-insurgency -0.01B each
	set_variable = { BRM_eth_money = BRM_eth_ceasefires }
	subtract_from_variable = { BRM_eth_money = BRM_eth_wars }
	multiply_variable = { BRM_eth_money = 0.01 }
}
