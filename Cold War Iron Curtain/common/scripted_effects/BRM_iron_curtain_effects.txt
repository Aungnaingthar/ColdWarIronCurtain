BRM_game_start_setup = {
	if = {
		limit = { NOT = { has_country_flag = BRM_focus_mod_setup } }
		set_country_flag = BRM_focus_mod_setup
		# Iron Curtain wipes the focus tree of countries without a 1960s tree in January 1960.
		# This flag makes Iron Curtain skip Burma, so the Burma tree stays.
		set_country_flag = focus_tree_60_reset
		# Iron Curtain's own Burma spirits have no pictures ("NO ICON?").
		# Swap them for identical copies that have pictures; the focus tree removes them.
		BRM_swap_iron_curtain_spirits = yes
		add_ideas = BRM_golden_land_heritage
		BRM_add_starting_leader_trait = yes
		BRM_recruit_advisors = yes
		if = {
			limit = { date > 1979.6.1 }
			BRM_setup_1980_start = yes
		}
	}
	# ethnic armed conflicts (Ethnic Affairs decisions)
	BRM_eth_init = yes
}

BRM_swap_iron_curtain_spirits = {

	if = { limit = { has_idea = BRM_Kuomintang_Tensions_2 } swap_ideas = { remove_idea = BRM_Kuomintang_Tensions_2 add_idea = BRM_kmt_tension_2 } }

}

################################################################
# LOANS BETWEEN COUNTRIES (money is Iron Curtain's currentMoney, in billions)
# 10-year loans, simple interest, repaid monthly (120 payments)
################################################################
BRM_send_loan_offer = {
	# Burma lends at 5% a year: total = amount x 1.5
	set_variable = { BRM_pending_rate = 5 }
	set_variable = { BRM_pending_total = BRM_pending_amount }
	multiply_variable = { BRM_pending_total = 1.5 }
	set_variable = { BRM_pending_payment = BRM_pending_total }
	divide_variable = { BRM_pending_payment = 120 }
	# the money leaves our treasury now (held until they answer; refunded if they refuse)
	subtract_from_variable = { currentMoney = BRM_pending_amount }
	set_country_flag = { flag = BRM_loan_pending days = 30 value = 1 }
	global_event_target:BRM_loan_partner = { country_event = { id = brmfin.2 days = 1 } }
}
BRM_send_loan_request = {
	# total = amount x (1 + rate/10)
	set_variable = { BRM_pending_total = BRM_pending_rate }
	divide_variable = { BRM_pending_total = 10 }
	add_to_variable = { BRM_pending_total = 1 }
	multiply_variable = { BRM_pending_total = BRM_pending_amount }
	set_variable = { BRM_pending_payment = BRM_pending_total }
	divide_variable = { BRM_pending_payment = 120 }
	set_country_flag = { flag = BRM_loan_pending days = 30 value = 1 }
	global_event_target:BRM_loan_partner = { country_event = { id = brmfin.11 days = 1 } }
}

BRM_process_loans = {
	set_variable = { BRM_total_receivable = 0 }
	set_variable = { BRM_total_payable = 0 }
	set_variable = { BRM_loan_income = 0 }
	set_variable = { BRM_loan_payments = 0 }
	# --- money other countries owe Burma ---
	every_country = {
		limit = { has_variable = BRM_owed_to_burma }
		if = {
			limit = { check_variable = { currentMoney > BRM_pay_to_burma } }
			subtract_from_variable = { currentMoney = BRM_pay_to_burma }
			subtract_from_variable = { BRM_owed_to_burma = BRM_pay_to_burma }
			ROOT = {
				add_to_variable = { currentMoney = PREV.BRM_pay_to_burma }
				add_to_variable = { BRM_loan_income = PREV.BRM_pay_to_burma }
			}
			set_variable = { BRM_missed_to_burma = 0 }
		}
		else = {
			add_to_variable = { BRM_missed_to_burma = 1 }
		}
		if = {
			limit = { check_variable = { BRM_missed_to_burma > 5 } }
			ROOT = { country_event = { id = brmfin.5 } }
			BRM_clear_loan_to_burma = yes
		}
		else_if = {
			limit = { check_variable = { BRM_owed_to_burma < 0.0005 } }
			ROOT = { country_event = { id = brmfin.6 } }
			BRM_clear_loan_to_burma = yes
		}
		else = {
			ROOT = { add_to_variable = { BRM_total_receivable = PREV.BRM_owed_to_burma } }
		}
	}
	# --- money Burma owes other countries ---
	every_country = {
		limit = { has_variable = BRM_burma_owes }
		if = {
			limit = { ROOT = { check_variable = { currentMoney > PREV.BRM_burma_pay } } }
			ROOT = {
				subtract_from_variable = { currentMoney = PREV.BRM_burma_pay }
				add_to_variable = { BRM_loan_payments = PREV.BRM_burma_pay }
			}
			add_to_variable = { currentMoney = BRM_burma_pay }
			subtract_from_variable = { BRM_burma_owes = BRM_burma_pay }
			set_variable = { BRM_burma_missed = 0 }
		}
		else = {
			add_to_variable = { BRM_burma_missed = 1 }
		}
		if = {
			limit = { check_variable = { BRM_burma_missed > 5 } }
			ROOT = { country_event = { id = brmfin.14 } }
			BRM_clear_loan_from_burma = yes
		}
		else_if = {
			limit = { check_variable = { BRM_burma_owes < 0.0005 } }
			ROOT = { country_event = { id = brmfin.15 } }
			BRM_clear_loan_from_burma = yes
		}
		else = {
			ROOT = { add_to_variable = { BRM_total_payable = PREV.BRM_burma_owes } }
		}
	}
}
BRM_clear_loan_to_burma = {
	clear_variable = BRM_owed_to_burma
	clear_variable = BRM_pay_to_burma
	clear_variable = BRM_missed_to_burma
	remove_opinion_modifier = { target = BRM modifier = BRM_loan_partner }
}
BRM_clear_loan_from_burma = {
	clear_variable = BRM_burma_owes
	clear_variable = BRM_burma_pay
	clear_variable = BRM_burma_missed
}

	# Iron Curtain's own export market (resources other countries buy from us).
	# Clean up agreements left in older saves.

		limit = { has_variable = BRM_trade_months }
		clear_variable = BRM_trade_months
	}

		limit = { has_idea = BRM_western_sanctions }
		subtract_from_variable = { currentMoney = 0.05 }
	}
}
