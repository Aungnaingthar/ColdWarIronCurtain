# Burma focus tree expansion: helper effects (ICX)

BRM_x_build_steel = {
	random_owned_controlled_state = {
		limit = { is_core_of = ROOT free_building_slots = { building = steel_factory size > 0 include_locked = yes } }
		add_extra_state_shared_building_slots = 1
		add_building_construction = { type = steel_factory level = 1 instant_build = yes }
	}
}

# Raise the development level of one core state by one step (same as Iron Curtain's state upgrade decision, but free)
BRM_x_upgrade_state = {
	custom_effect_tooltip = BRM_x_upgrade_state_tt
	hidden_effect = {
		random_owned_controlled_state = {
			limit = { is_core_of = ROOT BRM_x_can_upgrade = yes }
			Upgrade_State_effect = yes
		}
	}
}
